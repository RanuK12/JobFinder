def scrape_platform(platform_name, config, keywords, lang):
    search_url = f"{config['url']}{quote_plus(' '.join(keywords))}"

    try:
        # Use a random user-agent for the request
        headers = get_random_user_agent()
        response = requests.get(search_url, headers=headers, timeout=15)
        response.raise_for_status()
    except requests.exceptions.RequestException as e:
        print(f"Error fetching {platform_name}: {str(e)}")
        return

    soup = BeautifulSoup(response.text, 'html.parser')
    jobs = []

    for item in soup.select(config['selectors']['item']):
        try:
            # Obtiene la URL relativa del enlace
            relative_url = item.select_one(config['selectors']['url'])['href']

            # Construye la URL absoluta usando urljoin
            base_url = config.get('base_url', search_url)  # Usa la URL base de la plataforma si está disponible
            absolute_url = urljoin(base_url, relative_url)

            job = {
                'title': get_text(item, config['selectors']['title']),
                'company': get_text(item, config['selectors']['company']),
                'location': get_text(item, config['selectors']['location']),
                'url': absolute_url,  # Guarda la URL absoluta
                'platform': platform_name,
                'keywords_matched': [kw for kw in keywords if kw in item.text.lower()]
            }
            jobs.append(job)
        except Exception as e:
            print(f"Error parsing item: {str(e)}")

    return jobs

def get_text(soup, selector):
    element = soup.select_one(selector)
    return element.get_text(strip=True) if element else 'No especificado'

def extract_keywords(text, lang):
    keyword_dict = {

    }
    words = re.findall(r'\b[a-zA-ZáéíóúüñÁÉÍÓÚÜÑ]{4,}\b', text.lower())
    return [word for word in words if word in keyword_dict.get(lang,)]

from pdfminer.high_level import extract_text
from docx import Document

def parse_cv(filepath):
    try:
        if filepath.endswith('.pdf'):
            text = extract_text(filepath)
        elif filepath.endswith('.docx'):
            doc = Document(filepath)
            text = "\n".join([para.text for para in doc.paragraphs])
        else:
            raise ValueError("Formato no soportado")
        return text
    except Exception as e:
        print(f"Error parsing CV: {str(e)}")
        return ""

from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize
import re

def match_jobs(cv_text, jobs):
    cv_keywords = extract_keywords(cv_text)
    lang = detect_language(cv_text)
    matched_jobs = []
    
    for job in jobs:
        # Combine job title, company, and description for matching
        job_text = f"{job['title']} {job.get('company', '')} {job.get('description', '')}".lower()
        job_keywords = extract_keywords(job_text)
        
        # Find common keywords
        common = set(cv_keywords).intersection(job_keywords)
        
        # Calculate match score based on:
        # 1. Number of common keywords (weight: 0.4)
        # 2. Keyword position in title (weight: 0.3)
        # 3. Keyword frequency in description (weight: 0.3)
        score = 0.0
        
        # 1. Common keywords score
        if common:
            score += 0.4 * (len(common) / max(len(cv_keywords), len(job_keywords)))
            
        # 2. Title match score
        title_words = job['title'].lower().split()
        title_matches = sum(1 for word in title_words if word in cv_keywords)
        if title_words:
            score += 0.3 * (title_matches / len(title_words))
            
        # 3. Description match score
        if 'description' in job:
            desc_words = job['description'].lower().split()
            desc_matches = sum(1 for word in desc_words if word in cv_keywords)
            if desc_words:
                score += 0.3 * (desc_matches / len(desc_words))
        
        # Add job if score is above threshold
        if score >= 0.2:  # Lower threshold to be more inclusive
            matched_jobs.append({
                **job,
                'score': score,
                'reasons': [
                    f"{'Habilidades coincidentes' if lang == 'es' else 'Matching skills'}: {', '.join(common)}",
                    f"{'Coincidencia en título' if lang == 'es' else 'Title match'}: {title_matches} keywords",
                    f"{'Coincidencia en descripción' if lang == 'es' else 'Description match'}: {desc_matches if 'description' in job else 0} keywords"
                ]
            })
    
    # Sort by match score
    return sorted(matched_jobs, key=lambda x: x['score'], reverse=True)[:10]

def extract_keywords(text):
    # Keywords by category
    keywords = {
        'programming': ['python', 'java', 'javascript', 'react', 'node', 'programación', 'desarrollo'],
        'engineering': ['engineer', 'mechanical', 'civil', 'ingeniería', 'mecánica'],
        'healthcare': ['doctor', 'nurse', 'medical', 'médico', 'enfermería'],
        'logistics': ['logistics', 'supply chain', 'warehouse', 'logística', 'almacén'],
        'hospitality': ['hotel', 'restaurant', 'customer service', 'hotelería', 'restaurante'],
        'chef': ['chef', 'cooking', 'kitchen', 'cocina', 'gastronomía'],
        'remote': ['remote', 'teletrabajo', 'work from home'],
        'junior': ['junior', 'entry level', 'sin experiencia', 'primer empleo'],
        'senior': ['senior', 'sénior', 'experto', 'experiencia'],
        'management': ['manager', 'lead', 'gerente', 'líder'],
        'design': ['design', 'designer', 'diseño', 'diseñador'],
        'marketing': ['marketing', 'social media', 'digital', 'redes sociales'],
        'sales': ['sales', 'ventas', 'comercial'],
        'finance': ['finance', 'accounting', 'finanzas', 'contabilidad'],
        'education': ['teacher', 'professor', 'educación', 'profesor']
    }
    
    # Extract words from text
    words = re.findall(r'\b[a-zA-ZáéíóúüñÁÉÍÓÚÜÑ]+\b', text.lower())
    
    # Find matching keywords
    found_keywords = []
    for category, category_keywords in keywords.items():
        for keyword in category_keywords:
            if keyword in words:
                found_keywords.append(keyword)
                
    # If no keywords found, use most common words
    if not found_keywords:
        stop_words = set([
            'the', 'be', 'to', 'of', 'and', 'a', 'in', 'that', 'have', 'i',
            'it', 'for', 'not', 'on', 'with', 'he', 'as', 'you', 'do', 'at',
            'this', 'but', 'his', 'by', 'from', 'they', 'we', 'say', 'her', 'she',
            'or', 'an', 'will', 'my', 'one', 'all', 'would', 'there', 'their', 'what'
        ])
        
        word_freq = {}
        for word in words:
            if word not in stop_words and len(word) > 2:
                word_freq[word] = word_freq.get(word, 0) + 1
                
        found_keywords = [word for word, _ in sorted(word_freq.items(), key=lambda x: x[1], reverse=True)[:5]]
        
    return list(set(found_keywords))[:10]  # Return up to 10 unique keywords

def detect_language(text):
    return 'es' if re.search(r'\b(y|de|el|la)\b', text.lower()) else 'en'
